import { BrowserRouter, Routes, Route, Link, useLocation } from 'react-router-dom';
import { 
  Map as MapIcon, 
  BarChart3, 
  Scale, 
  Search, 
  Database, 
  Home as HomeIcon,
  Menu,
  X,
  UploadCloud
} from 'lucide-react';
import { useState } from 'react';

// Placeholders for page components
import Home from './pages/Home';
import InteractiveMap from './pages/InteractiveMap';
import Analytics from './pages/Analytics';
import Compare from './pages/Compare';
import DataSources from './pages/DataSources';
import DistrictSearch from './pages/DistrictSearch';
import Contribute from './pages/Contribute';
import { DepositProvider } from './context/DepositContext';

function Sidebar({ mobileOpen, setMobileOpen }: { mobileOpen: boolean, setMobileOpen: (s: boolean) => void }) {
  const location = useLocation();
  
  const navItems = [
    { path: '/', label: 'Home', icon: HomeIcon },
    { path: '/map', label: 'Manganese Map', icon: MapIcon },
    { path: '/search', label: 'District/Mine Search', icon: Search },
    { path: '/analytics', label: 'Analytics', icon: BarChart3 },
    { path: '/compare', label: 'Compare Areas', icon: Scale },
    { path: '/sources', label: 'Data Sources', icon: Database },
    { path: '/contribute', label: 'Contribute Data', icon: UploadCloud },
  ];

  return (
    <>
      {/* Mobile overlay */}
      {mobileOpen && (
        <div 
          className="fixed inset-0 bg-slate-900/50 z-40 lg:hidden" 
          onClick={() => setMobileOpen(false)}
        />
      )}
      
      {/* Sidebar */}
      <div className={`fixed inset-y-0 left-0 z-50 w-64 bg-slate-900 text-white transform transition-transform duration-200 ease-in-out lg:translate-x-0 lg:static lg:block ${mobileOpen ? 'translate-x-0' : '-translate-x-full'}`}>
        <div className="h-16 flex items-center justify-between px-4 border-b border-slate-800">
          <div className="font-bold text-lg text-orange-400 tracking-tight flex items-center gap-2">
            <MapIcon size={24} className="text-orange-500" />
            Mn Intelligence
          </div>
          <button className="lg:hidden text-slate-400 hover:text-white" onClick={() => setMobileOpen(false)}>
            <X size={20} />
          </button>
        </div>
        
        <div className="p-4">
          <div className="text-xs font-semibold text-slate-500 uppercase tracking-wider mb-4">
            Navigation
          </div>
          <nav className="space-y-1">
            {navItems.map((item) => {
              const isActive = location.pathname === item.path;
              const Icon = item.icon;
              return (
                <Link
                  key={item.path}
                  to={item.path}
                  onClick={() => setMobileOpen(false)}
                  className={`flex items-center gap-3 px-3 py-2 rounded-md text-sm font-medium transition-colors ${
                    isActive 
                      ? 'bg-orange-500/10 text-orange-400' 
                      : 'text-slate-300 hover:bg-slate-800 hover:text-white'
                  }`}
                >
                  <Icon size={18} className={isActive ? 'text-orange-400' : 'text-slate-400'} />
                  {item.label}
                </Link>
              );
            })}
          </nav>
        </div>
        
        <div className="absolute bottom-0 w-full p-4 border-t border-slate-800">
          <div className="bg-slate-800 rounded-md p-3 text-xs text-slate-400">
            <p className="font-semibold text-orange-400 mb-1">PROTOTYPE</p>
            <p>DEMO DATA – NOT FOR INDUSTRIAL DECISION-MAKING.</p>
          </div>
        </div>
      </div>
    </>
  );
}

function Layout({ children }: { children: React.ReactNode }) {
  const [mobileOpen, setMobileOpen] = useState(false);
  
  return (
    <div className="flex h-screen bg-slate-50 overflow-hidden">
      <Sidebar mobileOpen={mobileOpen} setMobileOpen={setMobileOpen} />
      
      <div className="flex-1 flex flex-col min-w-0 overflow-hidden">
        {/* Top header for mobile */}
        <header className="lg:hidden h-16 flex items-center px-4 bg-white border-b border-slate-200">
          <button 
            onClick={() => setMobileOpen(true)}
            className="p-2 -ml-2 text-slate-500 hover:bg-slate-100 rounded-md"
          >
            <Menu size={24} />
          </button>
          <span className="ml-2 font-semibold text-slate-900">Mn Intelligence</span>
        </header>
        
        <main className="flex-1 overflow-y-auto">
          {children}
        </main>
      </div>
    </div>
  );
}

function App() {
  return (
    <DepositProvider>
      <BrowserRouter>
        <Layout>
          <Routes>
            <Route path="/" element={<Home />} />
            <Route path="/map" element={<InteractiveMap />} />
            <Route path="/search" element={<DistrictSearch />} />
            <Route path="/analytics" element={<Analytics />} />
            <Route path="/compare" element={<Compare />} />
            <Route path="/sources" element={<DataSources />} />
            <Route path="/contribute" element={<Contribute />} />
          </Routes>
        </Layout>
      </BrowserRouter>
    </DepositProvider>
  );
}

export default App;
