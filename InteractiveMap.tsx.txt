import { useState, useMemo } from 'react';
import { MapContainer, TileLayer, CircleMarker, Popup, LayersControl } from 'react-leaflet';
import { getColorForGrade, GradeCategory } from '../data/mockData';
import { Filter, Layers as LayersIcon, MapPin } from 'lucide-react';
import { useDeposits } from '../context/DepositContext';

export default function InteractiveMap() {
  const { deposits } = useDeposits();
  const [activeLayer, setActiveLayer] = useState<'quality' | 'quantity'>('quality');
  const [selectedGrade, setSelectedGrade] = useState<GradeCategory | 'All'>('All');

  const filteredDeposits = useMemo(() => {
    if (selectedGrade === 'All') return deposits;
    return deposits.filter(d => d.gradeCategory === selectedGrade);
  }, [selectedGrade, deposits]);

  return (
    <div className="h-full flex flex-col">
      <div className="bg-white px-4 py-3 border-b border-slate-200 flex flex-wrap items-center justify-between gap-4">
        <h2 className="text-lg font-semibold text-slate-800 flex items-center gap-2">
          <MapPin size={20} className="text-orange-500" />
          Interactive Manganese Map
        </h2>
        
        <div className="flex flex-wrap items-center gap-4">
          <div className="flex items-center gap-2 bg-slate-100 p-1 rounded-lg">
            <LayersIcon size={16} className="text-slate-500 ml-2" />
            <span className="text-sm font-medium text-slate-600 mr-2">Layer:</span>
            <button 
              onClick={() => setActiveLayer('quality')}
              className={`px-3 py-1 text-sm rounded-md transition-colors ${activeLayer === 'quality' ? 'bg-white shadow-sm font-semibold text-slate-900' : 'text-slate-600 hover:bg-slate-200'}`}
            >
              Quality (Grade)
            </button>
            <button 
              onClick={() => setActiveLayer('quantity')}
              className={`px-3 py-1 text-sm rounded-md transition-colors ${activeLayer === 'quantity' ? 'bg-white shadow-sm font-semibold text-slate-900' : 'text-slate-600 hover:bg-slate-200'}`}
            >
              Quantity (Resource)
            </button>
          </div>
          
          <div className="flex items-center gap-2 bg-slate-100 p-1 rounded-lg">
            <Filter size={16} className="text-slate-500 ml-2" />
            <select 
              value={selectedGrade}
              onChange={(e) => setSelectedGrade(e.target.value as any)}
              className="bg-transparent border-none text-sm font-medium text-slate-700 focus:ring-0 cursor-pointer pl-1 pr-6 py-1"
            >
              <option value="All">All Grades</option>
              <option value="Low">Low (&lt;20%)</option>
              <option value="Medium">Medium (20-30%)</option>
              <option value="High">High (30-40%)</option>
              <option value="Very High">Very High (&gt;40%)</option>
            </select>
          </div>
        </div>
      </div>

      <div className="flex-1 relative z-0">
        <MapContainer center={[22.0, 79.0]} zoom={5} className="h-full w-full">
          <LayersControl position="topright">
            <LayersControl.BaseLayer checked name="OpenStreetMap">
              <TileLayer
                attribution='&copy; <a href="https://www.openstreetmap.org/copyright">OSM</a>'
                url="https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png"
              />
            </LayersControl.BaseLayer>
            <LayersControl.BaseLayer name="Satellite">
              <TileLayer
                attribution='Tiles &copy; Esri'
                url="https://server.arcgisonline.com/ArcGIS/rest/services/World_Imagery/MapServer/tile/{z}/{y}/{x}"
              />
            </LayersControl.BaseLayer>
          </LayersControl>

          {filteredDeposits.map(deposit => {
            // Determine size based on quantity if that layer is active
            const radius = activeLayer === 'quantity' 
              ? Math.max(8, Math.min(24, deposit.resourceTotal * 0.5))
              : 10;
              
            // Determine color based on quality if that layer is active, else a generic blue
            const color = activeLayer === 'quality' 
              ? getColorForGrade(deposit.gradeCategory)
              : '#3b82f6';

            return (
              <CircleMarker
                key={deposit.id}
                center={deposit.coordinates}
                radius={radius}
                pathOptions={{
                  fillColor: color,
                  fillOpacity: 0.7,
                  color: '#ffffff',
                  weight: 2,
                }}
              >
                <Popup>
                  <div className="p-4 w-[300px]">
                    <div className="mb-3">
                      <h3 className="font-bold text-lg text-slate-900 leading-tight">{deposit.name}</h3>
                      <p className="text-sm text-slate-500">{deposit.district}, {deposit.state}</p>
                    </div>
                    
                    <div className="grid grid-cols-2 gap-y-2 gap-x-4 mb-4">
                      <div>
                        <p className="text-xs text-slate-500 uppercase">Grade</p>
                        <p className="font-semibold text-slate-800">{deposit.grade}% Mn</p>
                        <span className={`text-[10px] px-1.5 py-0.5 rounded font-medium inline-block mt-0.5`}
                          style={{ backgroundColor: `${getColorForGrade(deposit.gradeCategory)}33`, color: getColorForGrade(deposit.gradeCategory) }}
                        >
                          {deposit.gradeCategory}
                        </span>
                      </div>
                      <div>
                        <p className="text-xs text-slate-500 uppercase">Status</p>
                        <p className="font-semibold text-slate-800">{deposit.status}</p>
                      </div>
                      <div>
                        <p className="text-xs text-slate-500 uppercase">Resources</p>
                        <p className="font-semibold text-slate-800">{deposit.resourceTotal} MT</p>
                      </div>
                      <div>
                        <p className="text-xs text-slate-500 uppercase">Reserves</p>
                        <p className="font-semibold text-slate-800">{deposit.reserveProven} MT</p>
                      </div>
                    </div>
                    
                    <div className="border-t border-slate-100 pt-3 mb-3">
                      <p className="text-xs text-slate-500 uppercase mb-1">Infrastructure</p>
                      <div className="flex justify-between text-sm">
                        <span>Rail: {deposit.distanceToRail}km</span>
                        <span>Port: {deposit.distanceToPort}km</span>
                      </div>
                    </div>
                    
                    <div className="bg-slate-50 p-2 rounded text-xs text-slate-600">
                      <div className="flex justify-between mb-1">
                        <span className="font-medium">Confidence:</span>
                        <span className={
                          deposit.confidence === 'High' ? 'text-green-600 font-semibold' :
                          deposit.confidence === 'Medium' ? 'text-yellow-600 font-semibold' : 'text-red-500 font-semibold'
                        }>{deposit.confidence}</span>
                      </div>
                      <div className="flex justify-between">
                        <span className="font-medium">Updated:</span>
                        <span>{deposit.lastUpdated}</span>
                      </div>
                    </div>
                  </div>
                </Popup>
              </CircleMarker>
            )
          })}
        </MapContainer>

        {/* Legend */}
        <div className="absolute bottom-6 right-6 bg-white/90 backdrop-blur-sm p-4 rounded-lg shadow-lg border border-slate-200 z-[1000] max-w-[200px]">
          <h4 className="font-semibold text-sm text-slate-800 mb-2 border-b border-slate-200 pb-1">Legend</h4>
          
          {activeLayer === 'quality' && (
            <div className="space-y-2">
              <p className="text-xs text-slate-500 font-medium mb-1">Manganese Grade</p>
              <div className="flex items-center gap-2">
                <div className="w-3 h-3 rounded-full bg-red-500"></div>
                <span className="text-xs text-slate-700">Very High (&gt;40%)</span>
              </div>
              <div className="flex items-center gap-2">
                <div className="w-3 h-3 rounded-full bg-orange-500"></div>
                <span className="text-xs text-slate-700">High (30-40%)</span>
              </div>
              <div className="flex items-center gap-2">
                <div className="w-3 h-3 rounded-full bg-yellow-500"></div>
                <span className="text-xs text-slate-700">Medium (20-30%)</span>
              </div>
              <div className="flex items-center gap-2">
                <div className="w-3 h-3 rounded-full bg-green-500"></div>
                <span className="text-xs text-slate-700">Low (&lt;20%)</span>
              </div>
            </div>
          )}
          
          {activeLayer === 'quantity' && (
            <div className="space-y-3">
              <p className="text-xs text-slate-500 font-medium mb-1">Resource Size (MT)</p>
              <div className="flex items-center gap-3">
                <div className="w-6 h-6 rounded-full bg-blue-500 opacity-70 border-2 border-white flex-shrink-0"></div>
                <span className="text-xs text-slate-700">&gt;20 MT</span>
              </div>
              <div className="flex items-center gap-3">
                <div className="w-4 h-4 rounded-full bg-blue-500 opacity-70 border-2 border-white flex-shrink-0 ml-1"></div>
                <span className="text-xs text-slate-700">10-20 MT</span>
              </div>
              <div className="flex items-center gap-3">
                <div className="w-2 h-2 rounded-full bg-blue-500 opacity-70 border border-white flex-shrink-0 ml-2"></div>
                <span className="text-xs text-slate-700">&lt;10 MT</span>
              </div>
            </div>
          )}
        </div>
      </div>
    </div>
  );
}
