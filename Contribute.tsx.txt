import { useState } from 'react';
import { useDeposits } from '../context/DepositContext';
import { UploadCloud, CheckCircle2, AlertCircle } from 'lucide-react';
import { useNavigate } from 'react-router-dom';

export default function Contribute() {
  const { addDeposit } = useDeposits();
  const navigate = useNavigate();
  
  const [submitted, setSubmitted] = useState(false);
  const [formData, setFormData] = useState({
    name: '',
    state: '',
    district: '',
    lat: '',
    lng: '',
    grade: '',
    resourceTotal: '',
    dataSource: ''
  });

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    
    // Add to context (which saves to localStorage)
    addDeposit({
      name: formData.name,
      state: formData.state,
      district: formData.district,
      coordinates: [Number(formData.lat), Number(formData.lng)],
      grade: Number(formData.grade),
      resourceTotal: Number(formData.resourceTotal),
      dataSource: formData.dataSource || 'Crowdsourced Public Submission',
    });
    
    setSubmitted(true);
    
    // Reset form after short delay and redirect to map
    setTimeout(() => {
      navigate('/map');
    }, 2500);
  };

  if (submitted) {
    return (
      <div className="flex flex-col items-center justify-center h-full p-8 text-center">
        <CheckCircle2 size={64} className="text-green-500 mb-4" />
        <h2 className="text-2xl font-bold text-slate-900 mb-2">Data Submitted Successfully!</h2>
        <p className="text-slate-600 max-w-md mb-6">
          Thank you for contributing. Your deposit has been added to our database with a "Low Confidence" tag pending official verification. 
          It will now appear on the Interactive Map and Analytics dashboard.
        </p>
        <p className="text-sm text-slate-400">Redirecting to Map...</p>
      </div>
    );
  }

  return (
    <div className="p-6 lg:p-8 max-w-3xl mx-auto h-full overflow-y-auto">
      <div className="mb-6">
        <h1 className="text-2xl font-bold text-slate-900 flex items-center gap-2">
          <UploadCloud className="text-orange-500" />
          Contribute Manganese Data
        </h1>
        <p className="text-slate-600 mt-2">
          Help us build a more comprehensive geological map. Submit local or recently surveyed manganese deposit information below.
        </p>
      </div>

      <div className="bg-blue-50 border border-blue-100 p-4 rounded-xl mb-6 flex items-start gap-3 text-blue-800 text-sm">
        <AlertCircle className="flex-shrink-0 mt-0.5" size={18} />
        <p>
          <strong>Note on Data Confidence:</strong> Publicly crowdsourced data will be automatically tagged with 
          <span className="font-semibold text-red-600 mx-1">Low Confidence</span> 
          until verified by an administrator against official reports (e.g., GSI or IBM).
        </p>
      </div>

      <form onSubmit={handleSubmit} className="bg-white rounded-xl shadow-sm border border-slate-200 p-6">
        <div className="space-y-6">
          
          <div>
            <h3 className="text-lg font-semibold text-slate-800 mb-3 border-b border-slate-100 pb-2">Location Details</h3>
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-1">Deposit/Mine Name *</label>
                <input required type="text" value={formData.name} onChange={e => setFormData({...formData, name: e.target.value})} className="w-full px-3 py-2 border border-slate-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-orange-500/20 focus:border-orange-500" placeholder="e.g. New Balaghat Extension" />
              </div>
              <div className="grid grid-cols-2 gap-4">
                <div>
                  <label className="block text-sm font-medium text-slate-700 mb-1">State *</label>
                  <input required type="text" value={formData.state} onChange={e => setFormData({...formData, state: e.target.value})} className="w-full px-3 py-2 border border-slate-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-orange-500/20 focus:border-orange-500" placeholder="State" />
                </div>
                <div>
                  <label className="block text-sm font-medium text-slate-700 mb-1">District *</label>
                  <input required type="text" value={formData.district} onChange={e => setFormData({...formData, district: e.target.value})} className="w-full px-3 py-2 border border-slate-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-orange-500/20 focus:border-orange-500" placeholder="District" />
                </div>
              </div>
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-1">Latitude (Decimal) *</label>
                <input required type="number" step="any" value={formData.lat} onChange={e => setFormData({...formData, lat: e.target.value})} className="w-full px-3 py-2 border border-slate-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-orange-500/20 focus:border-orange-500" placeholder="e.g. 21.81" />
              </div>
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-1">Longitude (Decimal) *</label>
                <input required type="number" step="any" value={formData.lng} onChange={e => setFormData({...formData, lng: e.target.value})} className="w-full px-3 py-2 border border-slate-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-orange-500/20 focus:border-orange-500" placeholder="e.g. 80.18" />
              </div>
            </div>
          </div>

          <div>
            <h3 className="text-lg font-semibold text-slate-800 mb-3 border-b border-slate-100 pb-2">Geological Data</h3>
            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-1">Estimated Grade (% Mn) *</label>
                <input required type="number" step="0.1" min="0" max="100" value={formData.grade} onChange={e => setFormData({...formData, grade: e.target.value})} className="w-full px-3 py-2 border border-slate-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-orange-500/20 focus:border-orange-500" placeholder="e.g. 35.5" />
              </div>
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-1">Total Resource (Million Tonnes) *</label>
                <input required type="number" step="0.1" min="0" value={formData.resourceTotal} onChange={e => setFormData({...formData, resourceTotal: e.target.value})} className="w-full px-3 py-2 border border-slate-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-orange-500/20 focus:border-orange-500" placeholder="e.g. 12.5" />
              </div>
              <div className="md:col-span-2">
                <label className="block text-sm font-medium text-slate-700 mb-1">Data Source / Reference</label>
                <input type="text" value={formData.dataSource} onChange={e => setFormData({...formData, dataSource: e.target.value})} className="w-full px-3 py-2 border border-slate-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-orange-500/20 focus:border-orange-500" placeholder="e.g. Local Mining Survey 2024 (Optional)" />
              </div>
            </div>
          </div>

          <div className="pt-4 flex justify-end">
            <button type="submit" className="bg-orange-500 hover:bg-orange-600 text-white px-6 py-2.5 rounded-lg font-medium transition-colors flex items-center gap-2">
              <UploadCloud size={18} />
              Submit Deposit Data
            </button>
          </div>
        </div>
      </form>
    </div>
  );
}
